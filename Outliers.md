## What Exactly is an Outlier?

> “An observation which deviates so much from other observations as to arouse suspicions that it was generated by a different mechanism.”  
> — Hawkins (1980)

Modern definition (more pragmatic):

**An observation (or small group of observations) that appears surprising / unlikely given the majority of the data and the assumed data-generating process.**

## Taxonomy of Outliers

```text
Outliers
├── Global (point is extreme in the whole dataset)
├── Local (extreme compared to its neighborhood)
├──
├── Univariate
│   ├── Unimodal distribution assumption
│   └── Multimodal / mixture
├── Multivariate
│   ├── Linear correlation structure
│   └── Non-linear structure
└── Contextual / Conditional
    ├── Time-series (temporal context)
    ├── Spatial
    └── Group / conditional (same values normal in one segment, abnormal in another)
```

### 04_Zscore_vs_Modified_Zscore_Univariate.ipynb

```python
# Classical vs Modified (robust) Z-score

def modified_z_score(y):
    y = pd.Series(y)
    median_y = y.median()
    mad = (y - median_y).abs().median()
    return 0.6745 * (y - median_y) / mad if mad != 0 else np.zeros(len(y))

for col in num_cols:
    z_classic = (df[col] - df[col].mean()) / df[col].std()
    z_mod     = modified_z_score(df[col])
    
    plt.figure(figsize=(10,4))
    plt.subplot(121); sns.histplot(z_classic, kde=True); plt.title("Classical Z")
    plt.subplot(122); sns.histplot(z_mod,     kde=True); plt.title("Modified Z (MAD)")
    plt.suptitle(f"{col} – Z-score comparison"); plt.tight_layout(); plt.show()
    
    print(f"{col} – points |Z| > 3.5:")
    print("  classic :", df[col][z_classic.abs() > 3.5].index.tolist())
    print("  modified:", df[col][z_mod.abs()     > 3.5].index.tolist())
```

```python
# IQR method – mild (1.5×) vs extreme (3×)

Q1 = df[num_cols].quantile(0.25)
Q3 = df[num_cols].quantile(0.75)
IQR = Q3 - Q1

lower_mild = Q1 - 1.5 * IQR
upper_mild = Q3 + 1.5 * IQR
lower_ext  = Q1 - 3.0 * IQR
upper_ext  = Q3 + 3.0 * IQR

out_mild = ((df[num_cols] < lower_mild) | (df[num_cols] > upper_mild)).sum()
out_ext  = ((df[num_cols] < lower_ext)  | (df[num_cols] > upper_ext)).sum()

print(pd.DataFrame({"Mild outliers (1.5×)": out_mild, "Extreme (3×)": out_ext}))
```

### 06_MAD_Median_Absolute_Deviation_Robust_Univariate.ipynb

```python
# MAD – very robust scale

def detect_mad_outliers(series, thresh=3.5):
    median = series.median()
    mad = np.median(np.abs(series - median))
    if mad == 0: return np.zeros(len(series), dtype=bool)
    modified_z = 0.6745 * (series - median) / mad
    return modified_z.abs() > thresh

for col in num_cols:
    outliers = detect_mad_outliers(df[col])
    print(f"{col}: {outliers.sum()} outliers (MAD > {thresh})")
    
    plt.figure(figsize=(10,4))
    sns.histplot(df[col][~outliers], label="normal", color="C0")
    sns.histplot(df[col][ outliers], label="outlier", color="red")
    plt.title(f"MAD outlier detection – {col}"); plt.legend(); plt.show()
```

### 08_Mahalanobis_Distance_Multivariate_Gaussian.ipynb

```python
from scipy.stats import chi2

# Assume roughly Gaussian multivariate data
X = df[['feat1','feat2','feat3','feat4']].dropna()
X_std = StandardScaler().fit_transform(X)

cov = np.cov(X_std.T)
inv_cov = np.linalg.inv(cov)

mahal = np.sqrt(np.einsum('ij,jk,ik->i', X_std, inv_cov, X_std))

threshold = np.sqrt(chi2.ppf(0.975, df=X.shape[1]))   # ~97.5% quantile

plt.figure(figsize=(10,5))
sns.histplot(mahal, bins=60, kde=True)
plt.axvline(threshold, color='red', ls='--', label="97.5% threshold")
plt.title("Mahalanobis distances"); plt.legend(); plt.show()

outliers = mahal > threshold
print(f"Detected {outliers.sum()} multivariate outliers")
```

### 14_Autoencoder_Reconstruction_Error_Outlier_Detection.ipynb

```python
import torch
import torch.nn as nn

# Simple autoencoder
class Autoencoder(nn.Module):
    def __init__(self, input_dim):
        super().__init__()
        self.encoder = nn.Sequential(nn.Linear(input_dim, 32), nn.ReLU(), nn.Linear(32, 16))
        self.decoder = nn.Sequential(nn.Linear(16, 32), nn.ReLU(), nn.Linear(32, input_dim))
    def forward(self, x):
        return self.decoder(self.encoder(x))

model = Autoencoder(X_std.shape[1])
optimizer = torch.optim.Adam(model.parameters(), lr=0.01)
criterion = nn.MSELoss()

X_t = torch.tensor(X_std, dtype=torch.float32)

# Train
for epoch in range(100):
    optimizer.zero_grad()
    recon = model(X_t)
    loss = criterion(recon, X_t)
    loss.backward()
    optimizer.step()
    if epoch % 20 == 0: print(f"epoch {epoch} loss {loss.item():.5f}")

# Reconstruction error
with torch.no_grad():
    recon = model(X_t)
    errors = ((X_t - recon)**2).mean(dim=1).numpy()

sns.histplot(errors, bins=80, kde=True)
plt.title("Autoencoder reconstruction MSE"); plt.show()

threshold = np.quantile(errors, 0.97)
print(f"Outliers (error > {threshold:.4f}): {(errors > threshold).sum()}")
```

```python
import numpy as np
from sklearn.model_selection import cross_val_score, KFold
from sklearn.ensemble import RandomForestRegressor, RandomForestClassifier

import seaborn as sns
import matplotlib.pyplot as plt

# ─── Pick your dataset ─── (change according to your repo)
# Example A: regression → house prices, medical costs, etc.
# Example B: classification → fraud detection, churn, etc.

df = pd.read_csv("your_dataset.csv")
target = "your_target_column"           # e.g. "SalePrice", "fraud", "churn"
problem_type = "regression"             # or "classification"

X = df.drop(columns=[target]).select_dtypes("number").fillna(0)
y = df[target]

if problem_type == "classification":
    model = RandomForestClassifier(n_estimators=120, random_state=42, n_jobs=-1)
    scoring = "roc_auc"
else:
    model = RandomForestRegressor(n_estimators=120, random_state=42, n_jobs=-1)
    scoring = "neg_root_mean_squared_error"
```

```markdown
## Strategies to compare
```

```python
results = {}

# 1. Raw (no treatment)
cv = cross_val_score(model, X, y, cv=5, scoring=scoring, n_jobs=-1)
results["Raw"] = cv.mean()

# 2. IQR-based removal (mild)
Q1, Q3 = X.quantile(0.25), X.quantile(0.75)
IQR = Q3 - Q1
mask_iqr = ~((X < (Q1 - 1.5*IQR)) | (X > (Q3 + 1.5*IQR))).any(axis=1)
cv_iqr = cross_val_score(model, X[mask_iqr], y[mask_iqr], cv=5, scoring=scoring)
results["IQR remove (mild)"] = cv_iqr.mean()

# 3. Winsorize (cap) at 1% / 99%
from scipy.stats.mstats import winsorize
X_win = pd.DataFrame()
for col in X.columns:
    X_win[col] = winsorize(X[col], limits=[0.01, 0.01])

cv_win = cross_val_score(model, X_win, y, cv=5, scoring=scoring)
results["Winsorize 1%/99%"] = cv_win.mean()

# 4. Log-transform skewed features + keep all
skewed = X.skew().abs() > 1.0
X_log = X.copy()
for col in X.columns[skewed]:
    X_log[col] = np.log1p(X_log[col].clip(lower=0))

cv_log = cross_val_score(model, X_log, y, cv=5, scoring=scoring)
results["Log + clip skewed"] = cv_log.mean()

# 5. Isolation Forest removal (aggressive)
from sklearn.ensemble import IsolationForest
iso = IsolationForest(contamination=0.04, random_state=42)
outliers = iso.fit_predict(X) == -1
cv_iso = cross_val_score(model, X[~outliers], y[~outliers], cv=5, scoring=scoring)
results["Iso Forest remove ~4%"] = cv_iso.mean()
```

```python
pd.Series(results).sort_values(ascending=False if "neg_" in scoring else True).to_frame("CV score")
```

```markdown
## Interpretation patterns you will often see

- Regression (prices, counts): removing outliers → often improves RMSE  
- Classification (fraud, rare events): removing outliers → can **hurt** a lot (removes signal)  
- Winsorizing → usually safer than deletion  
- Log-transform → frequently helps more than removal  
```

```markdown
## Outlier Decision Cheat Sheet

| Situation                              | Recommended action                              | Risk level | Code snippet hint                           |
|----------------------------------------|--------------------------------------------------|------------|---------------------------------------------|
| Obvious measurement error              | Remove or correct                               | Low        | `df = df[df[col] < 999999]`                 |
| Genuine extreme (business interest)    | Keep as-is                                      | —          | —                                           |
| Strong skew, model hates it            | Log / sqrt / Box-Cox → or winsorize             | Low        | `np.log1p(df[col].clip(0))`                 |
| Many univariate outliers, weak signal  | Winsorize 1–5%                                  | Low        | `winsorize(s, [0.02,0.02])`                 |
| Multivariate strange after PCA/UMAP    | Isolation Forest or LOF → investigate           | Medium     | `IsolationForest(contamination=0.03)`       |
| Fraud / anomaly detection target       | Almost never remove                             | High       | Keep all + model with outlier-robust method |
| Time-series spike (sensor glitch)      | Interpolate or forward-fill                     | Medium     | `ts.interpolate()`                          |
| No clear domain rule                   | Cap at reasonable business limit                | Low        | `df['age'] = df['age'].clip(0, 100)`        |
```
